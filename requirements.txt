# ---------------------------------------------------------------------------
# Core runtime requirements.
#
# These exact pins are shared by BOTH the Flask API image (app/Dockerfile) and
# the Airflow image (docker-compose.yml). The Airflow container WRITES the
# Prophet model JSON that the Flask container READS, so a prophet version drift
# between the two images breaks the /reload endpoint. Do not bump prophet in
# one place only.
# ---------------------------------------------------------------------------

# --- Forecasting model ---
# Prophet ships a prebuilt Stan binary in its manylinux wheel, so no compiler
# is needed at image build time. See reports/ for the Prophet-vs-NeuralProphet
# image-size tradeoff writeup.
prophet==1.1.6
cmdstanpy==1.2.4
holidays==0.57

# --- Data / numerics ---
# numpy pinned <2 because prophet 1.1.6's bundled Stan extension and several
# yfinance/pandas paths are only validated against the 1.x ABI.
numpy==1.26.4
pandas==2.2.3
scikit-learn==1.5.2

# --- Data source ---
# yfinance rate-limits and intermittently returns empty frames; src/etl_fetch.py
# wraps it in retry + raw CSV caching.
yfinance==0.2.51

# --- API ---
Flask==3.0.3
gunicorn==22.0.0
requests==2.32.3
